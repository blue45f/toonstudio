function synthetic(kind: string, width: number, height: number): Uint8Array {
  const bytes = new Uint8Array(width * height * 4);
  for (let y = 0; y < height; y++) for (let x = 0; x < width; x++) {
    const i = (y * width + x) * 4;
    if (kind === 'checker') {
      const c = ((Math.floor(x / 13) + Math.floor(y / 13)) & 1) * 255;
      bytes.set([c, c, c, 255], i);
    } else if (kind === 'transparent') {
      // 완전히 투명한 녹색 옆의 불투명 빨강. alpha 아래의 RGB와 합성 색을 구분한다.
      bytes.set(x < width / 2 ? [255, 0, 0, 255] : [0, 255, 0, 0], i);
    } else {
      bytes.set([Math.round(x / (width - 1) * 255), Math.round(y / (height - 1) * 255),
        Math.round((x + y) / (width + height - 2) * 255), 255], i);
    }
  }
  return bytes;
}

function compare(a: Uint8Array, b: Uint8Array, width: number, height: number, mode: 'raw' | 'black' | 'white', edgeOnly = false) {
  const channels = mode === 'raw' ? 4 : 3;
  const max = Array(channels).fill(0), sum = Array(channels).fill(0), square = Array(channels).fill(0);
  let count = 0;
  const read = (bytes: Uint8Array, p: number, c: number) => mode === 'raw' ? bytes[p + c]
    : bytes[p + c] * bytes[p + 3] / 255 + (mode === 'white' ? 255 - bytes[p + 3] : 0);
  const left = edgeOnly ? Math.floor(width / 2) - 8 : 0;
  const right = edgeOnly ? Math.ceil(width / 2) + 8 : width;
  for (let y = 0; y < height; y++) for (let x = left; x < right; x++) {
    const p = (y * width + x) * 4; count++;
    for (let c = 0; c < channels; c++) {
      const e = Math.abs(read(a, p, c) - read(b, p, c));
      max[c] = Math.max(max[c], e); sum[c] += e; square[c] += e * e;
    }
  }
  const mse = square.reduce((a, b) => a + b, 0) / (count * channels);
  // 8×8 비중첩 luminance 창, 모집단 분산, C1=(.01*255)^2, C2=(.03*255)^2.
  let ssim = 0, windows = 0;
  for (let y = 0; y < height; y += 8) for (let x = left; x < right; x += 8) {
    let sa = 0, sb = 0, saa = 0, sbb = 0, sab = 0, n = 0;
    for (let yy = y; yy < Math.min(y + 8, height); yy++) for (let xx = x; xx < Math.min(x + 8, right); xx++) {
      const p = (yy * width + xx) * 4;
      const aa = .2126 * read(a, p, 0) + .7152 * read(a, p, 1) + .0722 * read(a, p, 2);
      const bb = .2126 * read(b, p, 0) + .7152 * read(b, p, 1) + .0722 * read(b, p, 2);
      sa += aa; sb += bb; saa += aa * aa; sbb += bb * bb; sab += aa * bb; n++;
    }
    const ma = sa / n, mb = sb / n, va = Math.max(0, saa / n - ma * ma), vb = Math.max(0, sbb / n - mb * mb), cov = sab / n - ma * mb;
    ssim += ((2 * ma * mb + 6.5025) * (2 * cov + 58.5225)) / ((ma * ma + mb * mb + 6.5025) * (va + vb + 58.5225)); windows++;
  }
  return { channels: mode === 'raw' ? ['R', 'G', 'B', 'A'] : ['R', 'G', 'B'], pixels: count,
    maxAbsByChannel: max, maeByChannel: sum.map(n => n / count), mae: sum.reduce((a,b)=>a+b,0)/(count*channels),
    mse, psnrDb: mse === 0 ? 'Infinity' : 10 * Math.log10(255 * 255 / mse), luminanceSsim8x8: ssim / windows };
}

function alphaSummary(bytes: Uint8Array, transparent: boolean) {
  let min = 255, max = 0, zero = 0, fractional = 0, hiddenRgb = 0, greenOnBlackMax = 0;
  for (let p = 0; p < bytes.length; p += 4) {
    const a = bytes[p + 3]; min = Math.min(min, a); max = Math.max(max, a);
    if (a === 0) { zero++; if(bytes[p] || bytes[p+1] || bytes[p+2]) hiddenRgb++; }
    else if (a < 255) fractional++;
    if (transparent) greenOnBlackMax = Math.max(greenOnBlackMax,bytes[p+1] * a / 255);
  }
  return { min, max, zeroPixels:zero, fractionalPixels:fractional, zeroAlphaWithHiddenRgbPixels:hiddenRgb,
    ...(transparent ? { greenOnBlackMax } : {}) };
}
