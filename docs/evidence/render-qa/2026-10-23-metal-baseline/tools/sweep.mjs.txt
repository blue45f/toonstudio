  });
  server.on("error", reject);
});

function listProcesses() {
  const text = execFileSync("ps", ["-axo", "pid=,ppid=,command="], { encoding: "utf8", maxBuffer: 32 * 1024 * 1024 });
  return text.split("\n").filter(Boolean).map((line) => {
    const match = line.trim().match(/^(\d+)\s+(\d+)\s+(.*)$/u);
    return match ? { pid: Number(match[1]), ppid: Number(match[2]), command: match[3] } : null;
  }).filter(Boolean);
}
const STRAY_PATTERN = /ms-playwright|playwright_chromiumdev_profile|\/ts-sweep\/|chrome-headless-shell/u;
function killStrays(before) {
  const killed = [];
  for (const proc of listProcesses()) {
    if (before.has(proc.pid) || proc.pid === process.pid || !STRAY_PATTERN.test(proc.command)) continue;
    try { process.kill(proc.pid, "SIGKILL"); killed.push(proc.pid); } catch { /* 이미 종료 */ }
  }
  return killed;
}
const killGroup = (pid, signal) => { try { process.kill(-pid, signal); } catch { /* 이미 종료 */ } };

function runProcess({ cmd, env, timeoutSec, logPath }) {
  return new Promise((resolve) => {
    const fd = openSync(logPath, "w");
    const started = Date.now();
    const child = spawn(cmd[0], cmd.slice(1), { cwd: TREE, env: { ...baseEnv, ...env }, stdio: ["ignore", fd, fd], detached: true });
    let timedOut = false;
    let settled = false;
    const finish = (result) => {
      if (settled) return;
      settled = true;
      clearTimeout(timer);
      closeSync(fd);
      resolve({ ...result, timedOut, durationSec: Math.round((Date.now() - started) / 100) / 10 });
    };
    const timer = setTimeout(() => {
      timedOut = true;
      killGroup(child.pid, "SIGTERM");
      setTimeout(() => killGroup(child.pid, "SIGKILL"), 10_000);
    }, timeoutSec * 1000);
    child.on("error", (error) => finish({ exitCode: null, signal: null, spawnError: String(error.message) }));
    child.on("exit", (code, signal) => finish({ exitCode: code, signal }));
  });
}

const redact = (text) => text
  .replaceAll(TREE, "<sweep-tree>")
  .replaceAll(OUT, "<sweep-out>")
  .replaceAll(process.env.HOME ?? "/nonexistent-home", "~");
const tailLines = (path, count) => {
  try { return redact(readFileSync(path, "utf8")).trimEnd().split("\n").slice(-count); } catch { return []; }
};

const ADAPTER_PATTERNS = [
  /ANGLE \((?:Apple|Google|Intel|NVIDIA|AMD|ATI|Microsoft)[^)]*\)[^"\\\n]{0,80}/gu,
  /SwiftShader[^"\\\n,]{0,60}/giu,
  /Apple M\d[^"\\\n,]{0,20}/gu,
  /"(?:renderer|unmaskedRenderer|rendererString|adapterDescription|description|architecture|vendor|gpuLane|lane|actualDevice)"\s*:\s*"[^"]{1,120}"/gu,
];
function collectAdapterEvidence(texts) {
  const found = new Set();
  for (const text of texts) {
    for (const pattern of ADAPTER_PATTERNS) {
      for (const match of text.matchAll(pattern)) {
        found.add(redact(match[0]).trim());
        if (found.size >= 24) return [...found];
      }
    }
  }
  return [...found];
}
function walkFiles(dir, limit = 400) {
  const files = [];
  const stack = [dir];
  while (stack.length > 0 && files.length < limit) {
    const current = stack.pop();
    let entries;
    try { entries = readdirSync(current, { withFileTypes: true }); } catch { continue; }
    for (const entry of entries) {
      const full = join(current, entry.name);
      if (entry.isDirectory()) stack.push(full); else files.push(full);
    }
  }
  return files;
}
function artifactSummary(dir) {
  if (!existsSync(dir)) return { files: [], jsonTexts: [] };
  const files = walkFiles(dir);
  const jsonTexts = [];
  for (const file of files) {
    if (!file.endsWith(".json")) continue;
    try { if (statSync(file).size <= 2_000_000) jsonTexts.push(readFileSync(file, "utf8")); } catch { /* 건너뜀 */ }
  }
  return { files: files.map((file) => relative(dir, file)).sort(), jsonTexts };
}

async function startPreview() {
  const port = await freePort();
  const logPath = join(LOGS, `preview-${port}.log`);
  const fd = openSync(logPath, "w");
  const child = spawn("pnpm", ["exec", "vite", "preview", "--config", "apps/web/vite.config.ts", "--host", "127.0.0.1", "--port", String(port), "--strictPort"],
    { cwd: TREE, env: baseEnv, stdio: ["ignore", fd, fd], detached: true });
  const origin = `http://127.0.0.1:${port}`;
  for (let attempt = 0; attempt < 60; attempt += 1) {
    const ok = await new Promise((resolve) => {
      const request = http.get(`${origin}/`, (response) => { response.resume(); resolve(response.statusCode === 200); });
      request.on("error", () => resolve(false));
      request.setTimeout(2000, () => { request.destroy(); resolve(false); });
    });
    if (ok) return { origin, stop: () => { killGroup(child.pid, "SIGTERM"); setTimeout(() => killGroup(child.pid, "SIGKILL"), 3000); closeSync(fd); } };
    await new Promise((resolve) => setTimeout(resolve, 1000));
  }
  killGroup(child.pid, "SIGKILL");
  throw new Error("vite preview 가 60초 안에 응답하지 않았다");
}

const results = existsSync(RESULTS_PATH) ? JSON.parse(readFileSync(RESULTS_PATH, "utf8")).gates ?? [] : [];
const saveResults = () => writeFileSync(RESULTS_PATH, `${JSON.stringify({
  generatedAt: new Date().toISOString(), platform: process.platform, arch: process.arch,
  treeHead: (() => { try { return execFileSync("git", ["rev-parse", "HEAD"], { cwd: TREE, encoding: "utf8" }).trim(); } catch { return null; } })(),
  gates: results, notRun: NOT_RUN,
}, null, 2)}\n`);

for (const gate of GATES) {
  if (ONLY.length > 0 && !ONLY.includes(gate.id)) continue;
  progress(`START ${gate.id}`);
  const before = new Set(listProcesses().map((proc) => proc.pid));
  const entry = { id: gate.id, script: gate.script, command: gate.cmd.join(" "), envOverrides: Object.keys(gate.env ?? {}), declaredLane: gate.declaredLane, notes: gate.notes ?? null };
  let preview = null;
  try {
    if (gate.prep) {
      const prep = await runProcess({ cmd: gate.prep, env: {}, timeoutSec: 600, logPath: join(LOGS, `${gate.id}.prep.log`) });
      entry.prep = { command: gate.prep.join(" "), exitCode: prep.exitCode, durationSec: prep.durationSec };
      if (prep.exitCode !== 0) { entry.outcome = "prep-failed"; }
    }
    if (entry.outcome !== "prep-failed") {
      const env = { ...(gate.env ?? {}) };
      if (gate.needsPreview) { preview = await startPreview(); env.TOONSPECTRUM_VERIFY_ORIGIN = preview.origin; }
      const logPath = join(LOGS, `${gate.id}.log`);
      const run = await runProcess({ cmd: gate.cmd, env, timeoutSec: gate.timeoutSec, logPath });
      entry.exitCode = run.exitCode;
      entry.signal = run.signal ?? null;
      entry.durationSec = run.durationSec;
      entry.timeoutSec = gate.timeoutSec;
      entry.outcome = run.spawnError ? "error" : run.timedOut ? "hang" : run.exitCode === 0 ? "pass" : "fail";
      const { files, jsonTexts } = artifactSummary(art(gate.id === "character-lab-browser-probe" ? "character-lab" : gate.id));
      entry.artifactFileCount = files.length;
      entry.artifactSample = files.slice(0, 12);
      entry.adapterEvidence = collectAdapterEvidence([readFileSync(logPath, "utf8"), ...jsonTexts]);
      entry.logTail = tailLines(logPath, 14);
    }
  } catch (error) {
    entry.outcome = "error";
    entry.error = redact(String(error?.message ?? error));
  } finally {
    preview?.stop();
    entry.strayProcessesKilled = killStrays(before).length;
  }
  const index = results.findIndex((existing) => existing.id === entry.id);
  if (index === -1) results.push(entry); else results[index] = entry;
  saveResults();
  progress(`END   ${gate.id} -> ${entry.outcome} (${entry.durationSec ?? "-"}s)`);
}
saveResults();
progress("SWEEP-DONE");
